cmake_minimum_required(VERSION 3.13)

project(bldc_tft_demo C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

# Include Pico SDK
include(pico_sdk_import.cmake)

pico_sdk_init()

# Create executable
add_executable(bldc_tft_demo
    bldc_tft_demo.cpp
    st7789_pico.cpp
)

# Include directories
target_include_directories(bldc_tft_demo PRIVATE ${CMAKE_CURRENT_LIST_DIR})

# Link libraries
target_link_libraries(bldc_tft_demo 
    pico_stdlib
    hardware_spi
    hardware_gpio
    hardware_pwm
    hardware_uart
)

# Enable USB/UART output
pico_enable_stdio_usb(bldc_tft_demo 1)
pico_enable_stdio_uart(bldc_tft_demo 0)

# Generate UF2 firmware
pico_add_extra_outputs(bldc_tft_demo)
