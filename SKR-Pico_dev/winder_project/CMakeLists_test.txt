# Minimal test - just step motors
cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)

project(test_motors_simple C CXX ASM)
set(PICO_BOARD pico CACHE STRING "Board type")

pico_sdk_init()

add_executable(test_motors_simple
    test_motors_simple.cpp
)

target_link_libraries(test_motors_simple
    pico_stdlib
    hardware_gpio
)

pico_enable_stdio_usb(test_motors_simple 1)
pico_enable_stdio_uart(test_motors_simple 0)
pico_add_extra_outputs(test_motors_simple)
